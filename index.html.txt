async function fetchAIPrediction() {
const loadingEl = document.getElementById('loading');
const resultsEl = document.getElementById('prediction-results');

loadingEl.style.display = 'block';
resultsEl.innerHTML = '';

// サンプルデータ（出走表とリアルタイムオッズ）
const requestData = {
racers: [
{ bracket: 1, player_class: 4, national_win_rate: 7.12, local_win_rate: 7.50, motor_2rate: 45.2, avg_st: 0.12, motor_diff: 5.0, st_diff: -0.03, odds: 2.2 },

};

try {

// ※ローカルテストの場合は 'http://192.168.x.x:8000/api/predict' に変更

const response = await fetch(API_URL, {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify(requestData)
});

if (!response.ok) throw new Error('サーバー応答エラー');

const data = await response.json();
loadingEl.style.display = 'none';

// 結果を描画
data.predictions.forEach(item => {
const isHot = item.expected_value >= 1.0;
const div = document.createElement('div');
div.className = 'result-item';
div.innerHTML = `
<div>
<strong>${item.bracket}号艇</strong>
<span class="badge ${isHot ? 'hot-badge' : ''}">${isHot ? '期待値◎' : '通常'}</span>

</div>
`;

});

</div>
`;

});
}
